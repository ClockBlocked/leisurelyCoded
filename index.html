player.js 
// Bootstrap the Luister audio engine
import { createPlayer } from '@luister/core';

const player = createPlayer({
  container: '#stage',
  autoplay: true,
  quality: 'lossless',
  onReady: ({ duration }) => {
    console.log(`Ready — ${duration}s`);
  },
});

player.on('trackchange', (track) => {
  document.title = `${track.title} · ${track.artist}`;
});
